#include <tunables/global>

# DRAFT for the CortenDesk app.
# The image does NOT run under s6 but under supervisord (root) with php-fpm, nginx,
# hbbs, hbbr and the wrapper /run.sh -> the s6 baseline from the template does not apply.
#
# Starts in COMPLAIN mode: violations are only logged, nothing is
# blocked. After the test run, evaluate the host log (lines "apparmor=ALLOWED"
# or "profile=\"...cortendesk\""), close the gaps, then remove "complain" from the
# flags (enforce mode).
# Syntax checked (apparmor_parser -QK), runtime behavior untested.

profile cortendesk flags=(attach_disconnected,mediate_deleted,complain) {
  #include <abstractions/base>

  # --- Capabilities ---------------------------------------------------------
  # supervisord (root) wechselt per setuid/setgid zu www-data; Entrypoint macht
  # chown/chmod auf /data; nginx bindet Ports; supervisord signalisiert Kinder.
  capability chown,
  capability dac_override,
  capability dac_read_search,
  capability fowner,
  capability kill,
  capability net_bind_service,
  capability setgid,
  capability setuid,

  # --- Netzwerk -------------------------------------------------------------
  # hbbs/hbbr: TCP + UDP (21116/udp); nginx/php-fpm: TCP und Unix-Sockets.
  network inet stream,
  network inet dgram,
  network inet6 stream,
  network inet6 dgram,
  network unix stream,
  network unix dgram,

  signal (send,receive) peer=cortendesk,

  # --- Ausfuehrbare Dateien -------------------------------------------------
  /run.sh                        ix,
  /entrypoint.sh                 ix,
  /bin/**                        ix,
  /sbin/**                       ix,
  /usr/bin/**                    ix,
  /usr/sbin/**                   ix,
  /usr/local/bin/**              ix,
  /usr/local/sbin/**             ix,

  # --- Bibliotheken / Laufzeit (PHP, Python fuer supervisord, nginx-Module) ---
  /lib/**                        mr,
  /usr/lib/**                    mr,
  /usr/local/lib/**              mr,
  /usr/share/**                  r,
  /usr/local/include/**          r,

  # --- Anwendung ------------------------------------------------------------
  /app/**                        r,
  /app/storage/**                rwk,
  /app/bootstrap/cache/**        rwk,

  # --- Persistente Daten (SQLite, Schluessel, hbbs-DB, Uploads) --------------
  /data/                         r,
  /data/**                       rwk,

  # --- Configuration that the entrypoint writes at runtime ----------------
  /etc/**                        r,
  /etc/nginx/**                  rw,
  /etc/supervisor.d/**           rw,
  /usr/local/etc/php/**          rw,

  # --- Laufzeitverzeichnisse (PID-Dateien, Sockets, Logs, Temp) ---------------
  /run/**                        rwk,
  /var/run/**                    rwk,
  /var/lib/nginx/**              rwk,
  /var/log/**                    rw,
  /tmp/**                        rwk,
  /var/tmp/**                    rwk,

  # --- System-Info, die PHP/nginx/Python lesen --------------------------------
  /proc/**                       r,
  /sys/fs/cgroup/**              r,
  /dev/null                      rw,
  /dev/zero                      r,
  /dev/urandom                   r,
  /dev/stdout                    rw,
  /dev/stderr                    rw,
  /dev/pts/*                     rw,
}
