#include <tunables/global>

# NOTE: written from reading the init script and the three s6 services and from the processes seen in a
# Docker test (strace was not used). Enforce mode cannot be tested in Docker (§8): before relying on this
# profile run it once in complain mode on the Pi (apparmor="ALLOWED" in the journal), complete the rules,
# then remove complain again.
profile netbox flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # --- s6-overlay baseline (REQUIRED, §8) ---
  /init                          ix,
  /package/**                    ix,
  /command/**                    ix,
  /run/s6/**                     ix,
  /run/{,**}                     rwk,
  /etc/s6-overlay/**             ix,
  /etc/services.d/**             rwix,
  /etc/cont-init.d/**            rwix,
  /etc/cont-finish.d/**          rwix,

  # No /run.sh: startup runs through cont-init.d and s6-rc services, not through an entrypoint script.

  # Privilege drop (gosu to postgres, valkey and netbox), ownership of the data folders, stopping the services
  capability chown,
  capability dac_override,
  capability dac_read_search,
  capability fowner,
  capability setuid,
  capability setgid,
  capability kill,

  # --- Programs the scripts start ---
  /bin/**                        ix,
  /usr/bin/**                    ix,
  /usr/sbin/**                   ix,
  /usr/local/bin/**              ix,
  /usr/lib/bashio/**             rix,
  /etc/profile.d/**              r,

  # PostgreSQL server and client binaries (versioned directory), NetBox venv and source
  /usr/lib/postgresql/**         ix,
  /usr/share/postgresql/**       r,
  /opt/netbox/**                 rix,
  /usr/local/lib/netbox-ha/**    r,

  # Libraries and system data
  /lib/**                        mr,
  /usr/lib/**                    mr,
  /usr/share/**                  r,
  /etc/**                        r,
  /etc/ssl/certs/**              r,
  /etc/resolv.conf               r,
  /etc/hosts                     r,
  /etc/nsswitch.conf             r,

  # Persistent data: PostgreSQL cluster, Valkey dump, media, scripts, reports, secrets, plugins
  /data/                         rw,
  /data/**                       rwk,

  # App configuration folder (initial_admin_password.txt is written here, extra.py is read from here)
  /config/                       r,
  /config/**                     rw,

  # Scratch space (pg-init.log, granian.pid, PostgreSQL stats) and shared memory of PostgreSQL
  /tmp/                          rw,
  /tmp/**                        rwk,
  /dev/shm/                      rw,
  /dev/shm/**                    rwk,

  @{PROC}/**                     r,
  /sys/fs/cgroup/**              r,
  /sys/devices/system/cpu/**     r,
  /sys/kernel/mm/transparent_hugepage/** r,

  # Web UI, PostgreSQL and Valkey on 127.0.0.1, plugin installs from PyPI
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network unix stream,
  network unix dgram,
}
