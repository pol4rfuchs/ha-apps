#include <tunables/global>

profile matrix_whatsapp_bridge flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Name resolution + TLS certificates (homeserver, Whatsapp backend)
  /etc/resolv.conf               r,
  /etc/hosts                     r,
  /etc/nsswitch.conf             r,
  /etc/ssl/certs/**              r,
  /usr/share/ca-certificates/**  r,

  # ── s6-overlay baseline (required — otherwise crash loop on restart) ────────
  /init                          ix,
  /package/**                    ix,
  /command/**                    ix,
  /run/s6/**                     rix,
  /run/{,**}                     rwk,
  /etc/s6-overlay/**             rix,
  /etc/services.d/**             rwix,
  /etc/cont-init.d/**            rwix,
  /etc/cont-finish.d/**          rwix,

  # Own service scripts need ix, not just r
  /etc/s6-overlay/s6-rc.d/whatsapp_bridge/run    ix,
  /etc/s6-overlay/s6-rc.d/whatsapp_bridge/finish ix,

  # Shell, coreutils (bash, cp, mkdir, grep, awk, date, jq, yq, ...), bashio
  # and the s6 helpers (s6-envdir, importas, ... live in /usr/bin).
  # Determined with strace in the image: init + run script start exactly these tools.
  /bin/**                        ix,
  /usr/bin/**                    ix,
  /usr/sbin/**                   ix,
  /usr/local/bin/**              ix,
  /usr/lib/bashio/**             rix,
  /etc/profile.d/**              r,
  /dev/tty                       rw,

  # Example config (read-only source for the first start)
  /opt/mautrix-whatsapp/** r,

  # Data (SQLite DB, state store, config.yaml, registration.yaml)
  /data/ rw,
  /data/** rwk,

  # Shared volume for registration.yaml → Synapse
  /share/ r,
  /share/whatsapp_bridge_registration.yaml rw,

  # Options
  /data/options.json r,

  # Network
  network inet stream,
  network inet6 stream,
  network inet dgram,

  # Tmp
  /tmp/** rwk,

  deny /proc/sys/kernel/** w,
  deny @{PROC}/sysrq-trigger rwklx,
  deny @{PROC}/mem rwklx,
  deny @{PROC}/kmem rwklx,
  deny @{PROC}/kcore rwklx,
}
