#include <tunables/global>

profile forgejo_addon flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Network (HTTP, SSH, DNS)
  network udp,
  network tcp,
  network dgram,
  network stream,
  network inet,
  network inet6,
  network netlink raw,
  network unix dgram,
  network unix stream,

  # Capabilities for s6 and gosu
  capability setgid,
  capability setuid,
  capability dac_read_search,
  capability sys_chroot,

  # s6 overlay init system
  /init                     ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/**               ix,
  /command/**               ix,
  /run/{,**}                rwk,
  /dev/tty                  rw,

  # s6-overlay baseline (minimum per build rules §8): without these three paths
  # the container crashes on the first clean restart with
  # "/bin/sh: can't open '/init': Permission denied", because s6-overlay runs
  # these internal legacy-compat paths again itself during shutdown.
  /etc/services.d/**        rwix,
  /etc/cont-init.d/**       rwix,
  /etc/cont-finish.d/**     rwix,

  # Binaries
  /bin/**                   ix,
  /usr/bin/**               ix,
  /usr/sbin/**              ix,
  /usr/local/bin/**         ix,
  /app/gitea/**             ix,

  # bashio
  /usr/lib/bashio/**        ix,

  # s6 scripts
  /etc/s6/**                rix,
  /etc/s6-overlay/**        rix,
  /run/s6/**                rix,

  # System data
  /var/run/**               mrwkl,
  /tmp/**                   mrkwl,

  # Add-on data (addon_config mount)
  /config/**                rw,

  # SSL certificates (read-only)
  /ssl/**                   r,

  # Git user home
  /var/lib/gitea/**         rw,

  # Proc (for gosu / process information)
  @{PROC}/**                r,
  @{PROC}/*/fd/             rw,
  @{PROC}/*/status          r,

  # ptrace for Docker introspection
  ptrace (trace,read) peer=docker-default,

  # Signal from the Docker daemon
  signal (receive) set=(kill,term) peer=/usr/bin/docker,
}
