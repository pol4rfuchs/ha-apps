#include <tunables/global>

profile matrix_telegram_bridge flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Namensauflösung + TLS-Zertifikate (Homeserver, Telegram-Backend)
  /etc/resolv.conf               r,
  /etc/hosts                     r,
  /etc/nsswitch.conf             r,
  /etc/ssl/certs/**              r,
  /usr/share/ca-certificates/**  r,

  # ── s6-overlay Baseline (Pflicht — sonst Crash-Loop beim Restart) ────────
  /init                          ix,
  /package/**                    ix,
  /command/**                    ix,
  /run/s6/**                     rix,
  /run/{,**}                     rwk,
  /etc/s6-overlay/**             rix,
  /etc/services.d/**             rwix,
  /etc/cont-init.d/**            rwix,
  /etc/cont-finish.d/**          rwix,

  # Eigene Service-Scripts brauchen ix, nicht nur r
  /etc/s6-overlay/s6-rc.d/telegram_bridge/run    ix,
  /etc/s6-overlay/s6-rc.d/telegram_bridge/finish ix,

  # Shell, Coreutils (bash, cp, mkdir, grep, awk, date, jq, yq, ...), bashio
  # und die s6-Helfer (s6-envdir, importas, ... liegen in /usr/bin).
  # Per strace im Image ermittelt: Init + Run-Skript starten genau diese Tools.
  /bin/**                        ix,
  /usr/bin/**                    ix,
  /usr/sbin/**                   ix,
  /usr/local/bin/**              ix,
  /usr/lib/bashio/**             rix,
  /etc/profile.d/**              r,
  /dev/tty                       rw,

  # Beispiel-Config (Read-only-Quelle für Erststart)
  /opt/mautrix-telegram/** r,

  # Daten (SQLite-DB, State-Store, config.yaml, registration.yaml)
  /data/ rw,
  /data/** rwk,

  # Shared Volume für registration.yaml → Synapse
  /share/ r,
  /share/telegram_bridge_registration.yaml rw,

  # Optionen
  /data/options.json r,

  # Netzwerk
  network inet stream,
  network inet6 stream,
  network inet dgram,

  # Tmp
  /tmp/** rwk,

  deny /proc/sys/kernel/** w,
  deny @{PROC}/sysrq-trigger rwklx,
  deny @{PROC}/mem rwklx,
  deny @{PROC}/kmem rwklx,
  deny @{PROC}/kcore rwklx,
}
