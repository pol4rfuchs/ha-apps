#include <tunables/global>

# v3: Ausführen und Lesen nur auf die Pfade, die die App braucht; Schreiben, Fähigkeiten und Netzwerk eng.
profile ntfy_manager_addon flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Fähigkeiten: Docker-Standardsatz ohne net_raw
  capability chown,
  capability dac_override,
  capability dac_read_search,
  capability fowner,
  capability fsetid,
  capability kill,
  capability setgid,
  capability setuid,
  capability setpcap,
  capability net_bind_service,
  capability sys_chroot,
  capability mknod,
  capability audit_write,
  capability setfcap,

  # Netzwerk
  network tcp,
  network udp,
  network inet,
  network inet6,
  network unix,
  network netlink,
  deny network raw,
  deny network packet,
  deny mount,

  # Ausführen (s6-overlay, nginx, Node)
  /init                        mrix,
  /package/**                  mrix,
  /command/**                  mrix,
  /bin/**                      mrix,
  /sbin/**                     mrix,
  /usr/bin/**                  mrix,
  /usr/sbin/**                 mrix,
  /usr/local/bin/**            mrix,
  /usr/libexec/**              mrix,
  /usr/lib/bashio/**           mrix,
  /opt/ntfy-admin/**           mrix,
  /etc/s6-overlay/**           mrwix,
  /etc/services.d/**           mrwix,
  /etc/cont-init.d/**          mrwix,
  /etc/cont-finish.d/**        mrwix,
  /run/                        rw,
  /var/run/                    rw,
  /run/**                      mrwkix,
  /var/run/**                  mrwkix,
  /tmp/s6/**                   mrwkix,

  # Bibliotheken und Systemdateien (nur Lesen)
  /lib/**                      mr,
  /usr/lib/**                  mr,
  /usr/local/lib/**            mr,
  /etc/**                      r,
  /usr/share/**                r,
  /sys/**                      r,
  @{PROC}/**                   r,
  owner @{PROC}/@{pid}/**      rw,

  # Geräte
  /dev/null                    rw,
  /dev/zero                    rw,
  /dev/full                    rw,
  /dev/random                  r,
  /dev/urandom                 r,
  /dev/tty                     rw,
  /dev/ptmx                    rw,
  /dev/pts/**                  rw,
  /dev/shm/**                  rw,

  # Schreiben
  /tmp/                        rw,
  /tmp/**                      mrwk,
  /var/tmp/**                  rwk,
  /data/                       rw,
  /data/**                     rwk,
  /var/log/nginx/**            rw,
  /var/lib/nginx/**            rw,
  /var/cache/nginx/**          rw,

  # Signale und Prozess-Introspektion
  signal,
  ptrace (trace,read) peer=docker-default,
}
