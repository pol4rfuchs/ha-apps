#include <tunables/global>

# v3: Ausführen und Lesen nur auf die Pfade, die die App braucht; Schreiben, Fähigkeiten und Netzwerk eng.
profile teamspeak6_addon flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Fähigkeiten: Docker-Standardsatz ohne net_raw
  capability chown,
  capability dac_override,
  capability dac_read_search,
  capability fowner,
  capability fsetid,
  capability kill,
  capability setgid,
  capability setuid,
  capability setpcap,
  capability net_bind_service,
  capability sys_chroot,
  capability mknod,
  capability audit_write,
  capability setfcap,

  # Netzwerk
  network tcp,
  network udp,
  network inet,
  network inet6,
  network unix,
  network netlink,
  deny network raw,
  deny network packet,
  deny mount,

  # Ausführen (Debian-Image mit TeamSpeak-Server, bash, curl, jq, sqlite3)
  /bin/**                      mrix,
  /sbin/**                     mrix,
  /usr/bin/**                  mrix,
  /usr/sbin/**                 mrix,
  /usr/local/bin/**            mrix,
  /usr/libexec/**              mrix,
  /run.sh                      mrix,
  /opt/**                      mrix,
  /tsserver*                   mrix,

  # Bibliotheken und Systemdateien (nur Lesen)
  /lib/**                      mr,
  /lib64/**                    mr,
  /usr/lib/**                  mr,
  /usr/local/lib/**            mr,
  /etc/**                      r,
  /usr/share/**                r,
  /sys/**                      r,
  @{PROC}/**                   r,
  owner @{PROC}/@{pid}/**      rw,

  # Geräte
  /dev/null                    rw,
  /dev/zero                    rw,
  /dev/full                    rw,
  /dev/random                  r,
  /dev/urandom                 r,
  /dev/tty                     rw,
  /dev/pts/**                  rw,
  /dev/shm/**                  rw,

  # Schreiben
  /run/                        rw,
  /var/run/                    rw,
  /run/**                      rwk,
  /tmp/                        rw,
  /tmp/**                      rwk,
  /var/tmp/**                  rwk,
  /data/                       rw,
  /data/**                     rwk,
  /var/tsserver/               rw,
  /var/tsserver/**             rwk,

  # Signale und Prozess-Introspektion
  signal,
  ptrace (trace,read) peer=docker-default,
}
